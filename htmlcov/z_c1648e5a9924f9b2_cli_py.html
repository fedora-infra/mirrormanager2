142def main(ctx, config, debug, include_disabled, categories, startid, stopid, fraction, **kwargs): 

143 ctx.ensure_object(dict) 

144 ctx.obj["console"] = Console() 

145 

146 setup_logging(debug, ctx.obj["console"]) 

147 ctx.obj["options"] = ctx.params 

148 config = read_config(config) 

149 ctx.obj["config"] = config 

150 db_manager = get_db_manager(config) 

151 with db_manager.Session() as session: 

152 category_ids = [] 

153 for category_name in categories: 

154 category = get_category_by_name(session, category_name) 

155 if category is None: 

156 available_categories = "".join(c.name for c in get_categories(session)) 

157 raise click.BadOptionUsage( 

158 "--category", 

159 f"Category {category_name} does not exist. " 

160 f"Available categories: {available_categories}", 

161 ) 

162 category_ids.append(category.id) 

163 # Get *all* of the mirrors 

164 hosts = get_mirrors( 

165 session, 

166 private=False, 

167 order_by_crawl_duration=True, 

168 admin_active=True, 

169 user_active=None if include_disabled else True, 

170 site_private=False, 

171 site_user_active=True, 

172 site_admin_active=True, 

173 category_ids=category_ids or None, 

174 ) 

175 

176 # Limit our host list down to only the ones we really want to crawl 

177 if fraction and fraction != "1:1": 

178 if startid or stopid: 

179 raise click.BadOptionUsage( 

180 "--fraction", "Cannot use --fraction with --startid or --stopid" 

181 ) 

182 host_ids = [host.id for host in hosts] 

183 host_ids.sort() 

184 slices = int(fraction.split(":")[1]) 

185 part = int(fraction.split(":")[0]) 

186 start_index = (part - 1) * int(len(host_ids) / slices) 

187 stop_index = int(len(host_ids) / slices) * part 

188 

189 if slices == part: 

190 # Final part 

191 startid = host_ids[start_index] 

192 else: 

193 startid = host_ids[start_index] 

194 stopid = host_ids[stop_index] 

195 

196 ctx.obj["hosts"] = [ 

197 host for host in hosts if (host.id >= startid and (not stopid or host.id < stopid)) 

198 ] 

199 

200 # Before we do work, chdir to /var/tmp/. mirrormanager1 did this and I'm 

201 # not totally sure why... 

202 os.chdir("/var/tmp")

205def run_on_all_hosts(ctx_obj, options, report): 

206 logger.debug("Run with option: %s", repr(options)) 

207 starttime = time.monotonic() 

208 host_ids = [host.id for host in ctx_obj["hosts"]] 

209 results = [] 

210 error = None 

211 with Progress(console=ctx_obj["console"], refresh_per_second=1) as progress: 

212 task_global = progress.add_task(f"Crawling {len(host_ids)} mirrors", total=len(host_ids)) 

213 threads_results = run_in_threadpool( 

214 worker, 

215 host_ids, 

216 fn_args=(options, ctx_obj["config"], progress), 

217 timeout=options["global_timeout"], 

218 executor_kwargs={ 

219 "max_workers": options["threads"], 

220 }, 

221 ) 

222 try: 

223 for result in threads_results: 

224 progress.advance(task_global) 

225 if result is None: 

226 # The host has been skipped 

227 continue 

228 results.append(result) 

229 except GlobalTimeoutError as e: 

230 error = e 

231 

232 # Report what we have even if there was an error 

233 report(ctx_obj, options, results) 

234 duration = human_duration(time.monotonic() - starttime) 

235 if error is None: 

236 click.echo(f"Crawler finished after {duration}") 

237 elif options["no_fail"]: 

238 click.echo(f"Crawler failed after {duration}: {error}") 

239 else: 

240 raise click.ClickException(f"Crawler failed after {duration}: {error}")

294def propagation(ctx, **kwargs): 

295 """Records information about repomd.xml propagation and generates graphs. 

296 

297 Defaults to development/rawhide/x86_64/os/repodata 

298 """ 

299 options = ctx.obj["options"] 

300 options.update(ctx.params) 

301 options["propagation"] = True 

302 

303 # Set product_versions from product and version 

304 if options["version"] and not options["product"]: 

305 raise click.BadOptionUsage( 

306 "--version", "if you select a version, you must select a product." 

307 ) 

308 if not options["version"]: 

309 product_versions = get_current_versions(ctx.obj["config"]) 

310 if options["product"]: 

311 product_versions = [pv for pv in product_versions if pv[0] == options["product"]] 

312 else: 

313 product_versions = [(options["product"], options["version"])] 

314 del options["product"] 

315 del options["version"] 

316 options["product_versions"] = product_versions 

317 logger.info( 

318 "Propagation will be checked for %s", 

319 ", ".join(f"{pv[0]} {pv[1]}" for pv in product_versions), 

320 ) 

321 

322 # Only check propagation on active hosts 

323 ctx.obj["hosts"] = [host for host in ctx.obj["hosts"] if host.user_active] 

324 

325 run_on_all_hosts(ctx.obj, options, record_propagation)

328def record_propagation(ctx_obj, options, results: list[PropagationResult]): 

329 console = ctx_obj["console"] 

330 config = ctx_obj["config"] 

331 db_manager = get_db_manager(config) 

332 repo_status = defaultdict(lambda: defaultdict(lambda: 0)) 

333 for result in results: 

334 for repo_id, status in result.repo_status.items(): 

335 repo_status[repo_id][status.value] += 1 

336 now = datetime.now() 

337 with db_manager.Session() as session: 

338 for repo_id, status_counts in repo_status.items(): 

339 session.add(model.PropagationStat(repository_id=repo_id, datetime=now, **status_counts)) 

340 report_propagation(console, session, repo_status) 

341 session.commit()
