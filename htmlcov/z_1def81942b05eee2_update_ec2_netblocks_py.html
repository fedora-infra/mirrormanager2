2Synchronize Amazon AWS-published EC2 netblock lists per region into the

16def parse_out_region(hostname): 

17 if hostname.startswith("s3-mirror-"): 

18 hostname = hostname[len("s3-mirror-") :] 

19 if hostname.endswith(".fedoraproject.org"): 

20 hostname = hostname[: -len(".fedoraproject.org")] 

21 return hostname

24def s3_mirrors(session): 

25 hosts_by_region = {} 

26 site = mirrormanager2.lib.get_site_by_name(session, "Red Hat") 

27 for host in site.hosts: 

28 if host.name.startswith("s3"): 

29 region = parse_out_region(host.name) 

30 hosts_by_region[region] = { 

31 "host": host, 

32 "netblocks": {}, 

33 } 

34 for nb in host.netblocks: 

35 hosts_by_region[region]["netblocks"][nb.netblock] = { 

36 "host_netblock_id": nb.id, 

37 "stale": True, 

38 } 

39 return hosts_by_region

42def get_ip_ranges(): 

43 response = requests.get("https://ip-ranges.amazonaws.com/ip-ranges.json") 

44 response.raise_for_status() 

45 return response.json()

48def host_has_netblock(hosts_by_region, region, netblock): 

49 return netblock in hosts_by_region[region]["netblocks"] 

50

52def doit(session, dry_run): 

53 hosts_by_region = s3_mirrors(session) 

54 ipranges = get_ip_ranges() 

55 

56 for p in ipranges["prefixes"]: 

57 service = p["service"] 

58 region = p["region"] 

59 ip_prefix = p["ip_prefix"] 

60 if service != "EC2": 

61 continue 

62 if region == "GLOBAL": 

63 continue 

64 

65 if region in hosts_by_region: # ignore regions we don't have a mirror in 

66 h = hosts_by_region[region] 

67 host = h["host"] 

68 if not host_has_netblock(hosts_by_region, region, ip_prefix): 

69 print(f"Adding host {host.name} netblock {ip_prefix}") 

70 if not dry_run: 

71 nb = HostNetblock( 

72 host=host, netblock=ip_prefix, name=None 

73 ) # this adds the entry to the database, mark as not stale 

74 session.add(nb) 

75 session.flush() 

76 hosts_by_region[region]["netblocks"][ip_prefix] = { 

77 "host_netblock_id": nb.id, 

78 "stale": False, 

79 } 

80 else: 

81 # found the netblock in our database, mark it as not stale 

82 hosts_by_region[region]["netblocks"][ip_prefix]["stale"] = False 

83 

84 # delete stale netblock entries from the database 

85 for region, h in hosts_by_region.items(): 

86 for netblock in hosts_by_region[region]["netblocks"]: 

87 if hosts_by_region[region]["netblocks"][netblock][ 

88 "stale" 

89 ]: # delete this, it's no longer on Amazon's list 

90 host = h["host"] 

91 print(f"Deleting host {host.name} netblock {netblock}") 

92 if not dry_run: 

93 nb = mirrormanager2.lib.get_host_netblock( 

94 session, hosts_by_region[region]["netblocks"][netblock]["host_netblock_id"] 

95 ) 

96 session.delete(nb) 

97

99@click.command() 

100@config_option 

101@click.option("-n", "--dry-run", is_flag=True, default=False) 

102def main(config, dry_run): 

103 config = mirrormanager2.lib.read_config(config)
