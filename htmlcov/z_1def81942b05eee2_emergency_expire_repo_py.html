22""" 

23This script is an emergency script meant for releng to be used if an very 

24important update has went out, and we want to make sure that clients pick up 

25the latest version of the repo, even if that means more load on the master 

26mirrors. 

27It does this by deleting all non-current file_detail entries for this specific 

28repository, resulting in no alternate file checksums in the pickle. 

29This will cause the metalink to not return any <alternates> entries, such that 

30only the very latest version of the repo is considered as up-to-date. 

31""" 

32 

33import logging 

34 

35import click 

36 

37from mirrormanager2.lib import read_config 

38from mirrormanager2.lib.database import get_db_manager 

39from mirrormanager2.lib.model import Product, Repository, Version 

40 

41from .common import config_option 

42 

43logger = logging.getLogger(__name__) 

44 

45 

46def setup_logging(): 

47 format = "[%(asctime)s][%(name)10s %(levelname)7s] %(message)s" 

48 logging.basicConfig(format=format) 

49 

50 logger.setLevel(logging.INFO) 

51 

52 

53@click.command() 

54@config_option 

55@click.argument("product", help="Product to clear old filedetails for (Fedora, EPEL)") 

56@click.argument("version", help="VERSION to clear old filedetails for (20, 21)") 

57def main(config, product, version): 

58 config = read_config(config) 

59 

60 db_manager = get_db_manager(config) 

61 session = db_manager.Session() 

62 

63 setup_logging() 

64 

65 product = session.query(Product).filter_by(name=product).first() 

66 if not product: 

67 message = f"No product {product} found" 

68 logger.error(message) 

69 raise click.BadArgumentUsage(message) 

70 

71 version = session.query(Version).filter_by(name=version, product_id=product.id).first() 

72 if not version: 

73 message = f"No version {version} found" 

74 logger.error(message) 

75 raise click.BadArgumentUsage(message) 

76 

77 repos = session.query(Repository).filter_by(version_id=version.id).all() 

78 if len(repos) < 1: 

79 message = f"No repos found for {product} version {version}" 

80 logger.error(message) 

81 raise click.BadArgumentUsage(message) 

82 

83 for repo in repos: 

84 logger.info("Clearing for repo %s", repo.name) 

85 deleted = repo.emergency_expire_old_file_details(session) 

86 

87 if not deleted: 

88 logger.info("Unused repo") 

89 else: 

90 logger.info( 

91 "Older versions deleted: %s", 

92 ", ".join([filename for filename in deleted if deleted[filename] != 0]), 

93 ) 

94 

95 logger.info("Done.")
